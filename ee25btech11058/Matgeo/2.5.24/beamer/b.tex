\begin{frame}{Question}
If $|\vec{a}| = 2$, $|\vec{b}| = 7$ and $\vec{a} \times \vec{b} = 3\hat{\imath} + 2\hat{\jmath} + 6\hat{k}$, find the angle between $\vec{a}$ and $\vec{b}$.
\end{frame}

\begin{frame}{Theoretical Formula}
The magnitude of the cross product of two vectors is defined by the formula:
\begin{equation}
|\vec{a} \times \vec{b}| = |\vec{a}| |\vec{b}| \sin(\theta)
\label{eq:cross_product}
\end{equation}
To use this, we first need to calculate the magnitude of the given cross product vector.
\end{frame}

\begin{frame}{Solution using Matrices}
Let the vector $\vec{c} = \vec{a} \times \vec{b}$. We represent it as a column matrix:
\begin{equation}
\vec{c} = \myvec{3 \\ 2 \\ 6}
\end{equation}
The squared magnitude is found by $\vec{c}^\top\vec{c}$:
\begin{align*}
\norm{\vec{c}}^2 &= \myvec{3 & 2 & 6} \myvec{3 \\ 2 \\ 6} \\
&= (3 \cdot 3) + (2 \cdot 2) + (6 \cdot 6) = 9 + 4 + 36 = 49
\end{align*}
So, $|\vec{c}| = \sqrt{49} = 7$.
\end{frame}

\begin{frame}{Finding the Angle}
Now, substituting the values into Equation~\eqref{eq:cross_product}:
\begin{align*}
|\vec{a} \times \vec{b}| &= |\vec{a}| |\vec{b}| \sin(\theta) \\
7 &= (2)(7) \sin(\theta) \\
7 &= 14 \sin(\theta)
\end{align*}
Solving for $\sin(\theta)$:
\begin{equation*}
\sin(\theta) = \frac{7}{14} = \frac{1}{2}
\end{equation*}
Therefore, the angle is $\theta = \arcsin\left(\frac{1}{2}\right) = 30^\circ$ or $\frac{\pi}{6}$ radians.
\end{frame}


\begin{frame}[fragile]
    \frametitle{Solution 1: Pure C Code}
    \begin{lstlisting}
#include <stdio.h>
#include <math.h>

#ifndef M_PI
#define M_PI 3.14159265358979323846
#endif

double angle_between(double a, double b, double x, double y, double z) {
    double cross_mag = sqrt(x*x + y*y + z*z);
    double sin_theta = cross_mag / (a * b);
    double theta_rad = asin(sin_theta);
    return theta_rad * 180.0 / M_PI;
}

int main() {
    double angle = angle_between(2.0, 7.0, 3.0, 2.0, 6.0);
    printf("Angle between a and b = %.2f degrees\n", angle);
    return 0;
}
    \end{lstlisting}
\end{frame}
